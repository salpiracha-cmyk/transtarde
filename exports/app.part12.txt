
function expectedContainers(s,c){const n=(s.loading?.lots||[]).reduce((a,l)=>a+(l.allocations||[]).reduce((x,r)=>x+num(r.containers),0),0);return n||num(c.containers)}
function uniqueActualContainers(s){const seen=new Set();return(s.millActuals||[]).filter(a=>{const n=normalizeContainer(a.number);if(!CONTAINER_RE.test(n)||seen.has(n))return false;seen.add(n);return true})}
function blCompleteness(s,c){const rows=uniqueActualContainers(s),expected=expectedContainers(s,c),seals=rows.filter(a=>String(a.seal||'').trim()).length;return{expected,actual:rows.length,seals,complete:rows.length===expected&&seals===expected}}
function ensureBL(s,c){s.bl=Object.assign({draftSaved:false,finalized:false,blNo:'',onBoardDate:'',vessel:'',voyage:'',consignee:'TO ORDER',notify:c.buyerDetails?.name||s.buyer||'',description:s.customs?.description||c.product||'',hsCode:s.customs?.hsCode||'',finalFile:''},s.bl||{});return s.bl}

function captureBL(){const s=shipment(),c=contractByRef(s.contractRef),b=ensureBL(s,c);b.blNo=byId('blNo').value.trim();b.onBoardDate=byId('blDate').value;b.vessel=byId('blVessel').value.trim();b.voyage=byId('blVoyage').value.trim();b.consignee=byId('blConsignee').value.trim();b.notify=byId('blNotify').value.trim();b.hsCode=byId('blHS').value.trim();b.description=byId('blDescription').value.trim();return b}
function wireBL(){const s=shipment(),c=contractByRef(s.contractRef);byId('saveBLDraft').onclick=()=>{const b=captureBL();b.draftSaved=true;audit('B/L','Draft particulars saved',s.contractRef);save();renderShipmentWorkspace()};byId('finalizeBL').onclick=()=>{const b=captureBL(),z=blCompleteness(s,c);if(!z.complete)return alert('All expected containers and seals must be received from Mill before finalizing B/L.');if(!b.blNo||!b.onBoardDate||!b.vessel||!b.voyage)return alert('B/L Number, On Board Date, Vessel and Voyage are required for Final/Original B/L particulars.');b.finalized=true;b.finalizedAt=new Date().toISOString();s.status='B/L Finalized';audit('B/L','Final particulars confirmed',`${b.blNo} · ${b.vessel} ${b.voyage}`);save();printBL(s,c);renderShipmentWorkspace()}}

function actualLoadedQtyMT(s){return round((s.millActuals||[]).reduce((a,x)=>a+num(x.netKg),0)/1000,3)}
function ensureCommercial(s,c){s.commercial=Object.assign({saved:false,date:TODAY(),invoiceNo:c.ref||s.contractRef,rate:num(c.packings?.[0]?.price),currency:c.currency||'USD',gdRefs:[],advanceAllocations:[],bankInvoices:[],finalized:false},s.commercial||{});s.commercial.advanceAllocations=Array.isArray(s.commercial.advanceAllocations)?s.commercial.advanceAllocations:[];s.commercial.gdRefs=Array.isArray(s.commercial.gdRefs)?s.commercial.gdRefs:[];return s.commercial}
function commercialValue(s,c){const x=ensureCommercial(s,c);return round(actualLoadedQtyMT(s)*num(x.rate),2)}
function advanceDuplicate(ref,sid,index){if(!String(ref||'').trim())return false;const key=String(ref).trim().toUpperCase();for(const sh of state.shipments){for(let i=0;i<(sh.commercial?.advanceAllocations||[]).length;i++){const a=sh.commercial.advanceAllocations[i];if(String(a.ref||'').trim().toUpperCase()===key&&(sh.id!==sid||i!==index))return true}}return false}

function captureCommercial(){const s=shipment(),c=contractByRef(s.contractRef),x=ensureCommercial(s,c);x.invoiceNo=byId('comNo').value.trim();x.date=byId('comDate').value;x.currency=byId('comCurrency').value.trim().toUpperCase();x.rate=num(byId('comRate').value);x.gdRefs=(byId('comGD').value||'').split('\n').map(v=>v.trim()).filter(Boolean);document.querySelectorAll('[data-adv-ref]').forEach(el=>x.advanceAllocations[num(el.dataset.advRef)].ref=el.value.trim());document.querySelectorAll('[data-adv-amount]').forEach(el=>x.advanceAllocations[num(el.dataset.advAmount)].amount=num(el.value));return x}
function validateAdvanceAllocations(s,x){for(let i=0;i<x.advanceAllocations.length;i++){const a=x.advanceAllocations[i];if((a.ref&&!num(a.amount))||(!a.ref&&num(a.amount)))return'Every advance allocation requires both a reference and amount.';if(a.ref&&advanceDuplicate(a.ref,s.id,i))return`Advance / receipt reference ${a.ref} is already allocated to another invoice/lot.`}return''}
function wireCommercial(){const s=shipment(),c=contractByRef(s.contractRef),x=ensureCommercial(s,c);for(const id of ['comNo','comDate','comCurrency','comRate','comGD'])byId(id)?.addEventListener('change',()=>{captureCommercial();renderShipmentWorkspace()});byId('addAdvanceAllocation').onclick=()=>{captureCommercial();x.advanceAllocations.push({ref:'',amount:0});renderShipmentWorkspace()};document.querySelectorAll('[data-adv-remove]').forEach(el=>el.onclick=()=>{captureCommercial();x.advanceAllocations.splice(num(el.dataset.advRemove),1);renderShipmentWorkspace()});byId('saveCommercial').onclick=()=>{captureCommercial();const e=validateAdvanceAllocations(s,x);if(e)return alert(e);if(num(x.rate)<=0)return alert('Commercial Rate is required.');x.saved=true;audit('COMMERCIAL','Saved commercial invoice data',x.invoiceNo);save();renderShipmentWorkspace()};byId('finalCommercial').onclick=()=>{captureCommercial();const e=validateAdvanceAllocations(s,x),z=blCompleteness(s,c);if(e)return alert(e);if(num(x.rate)<=0)return alert('Commercial Rate is required.');if(!z.complete)return alert('Expected containers and valid seals must be complete.');if(!s.bl?.finalized||!s.bl?.vessel||!s.bl?.voyage)return alert('Final / Original B/L particulars and Vessel / Voyage are required.');if(!x.gdRefs.length)return alert('Complete GD information before Final Commercial Invoice.');x.saved=true;x.finalized=true;x.finalizedAt=new Date().toISOString();s.status='Commercial Documents Finalized';audit('COMMERCIAL','Final Commercial Invoice prepared',`${x.invoiceNo} · ${money(commercialValue(s,c),x.currency)}`);save();printCommercialInvoice(s,c);renderShipmentWorkspace()};byId('printCommPacking').onclick=()=>printCommercialPackingList(s,c)}
