Later rows may be TBA.');if(l.allocations.some(a=>!a.dateTBA&&!a.loadingDate))return alert('Enter Loading Date or mark that row TBA.');const total=l.allocations.reduce((a,x)=>a+allocationMT(x),0);if(total<=0)return alert('Lot quantity must be greater than zero.');if(total>remaining+0.001)return alert(`This lot exceeds the remaining Contract balance of ${remaining.toFixed(3)} MT.`);const allocations=l.allocations.map(a=>{const pk=loadingPack(c,a.packIndex);return{...clone(a),weightPer:num(pk.weightPer),brand:pk.brand||'',bagSize:num(pk.size),bagUnit:pk.unit||c.packingUnit||'KG',bagType:pk.type||'',bagsPerContainer:bagsPerContainer(pk,c),totalBags:bagsPerContainer(pk,c)*num(a.containers),dryOn:s.production.dryOn||'No',craftPaper:s.production.craftPaper||'No',dpp:s.production.dpp||'Yes',inspection:s.production.inspection||c.inspection||'None'}});const lot={lotId:l.lotId,totalMT:round(total,3),allocations,createdAt:new Date().toISOString(),productionSnapshot:clone(s.production)};s.loading.lots.push(lot);state.millSync.exportLoading=state.millSync.exportLoading.filter(x=>!(x.contractRef===s.contractRef&&x.lotId===lot.lotId));state.millSync.exportLoading.push({contractRef:s.contractRef,shipmentId:s.id,lotId:lot.lotId,totalMT:lot.totalMT,allocations:clone(allocations),production:clone(s.production),sentAt:lot.createdAt});s.loading.draft=null;s.status='Loading Instructions Sent';audit('LOADING INSTRUCTIONS','Created lot and sent instructions to Mill',`${s.contractRef} · ${lot.lotId} · ${lot.totalMT.toFixed(3)} MT`);save();view='home';currentShipmentId='';activeWorkspace='';renderHome()}
