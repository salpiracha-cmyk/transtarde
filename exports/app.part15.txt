
function printCOOCustomerDraft(s,c){const x=s.coo,w=customsWeightSummary(s,c);printEngine(`COO Draft ${x.reference||s.contractRef}`,`<h1 class="docTitle">CERTIFICATE OF ORIGIN — CUSTOMER DRAFT</h1><table class="meta"><tr><td class="lbl">Exporter</td><td>${esc(SELLERS[c.seller]?.name||c.seller)}<br>${esc(c.sellerDetails?.address||'')}</td><td class="lbl">Reference</td><td>${esc(x.reference||'')}</td></tr><tr><td class="lbl">Consignee</td><td>${esc(x.consignee||'')}</td><td class="lbl">Date</td><td>${fmt(x.date)}</td></tr><tr><td class="lbl">Vessel / Voyage</td><td>${esc(s.bl?.vessel||'')} / ${esc(s.bl?.voyage||'')}</td><td class="lbl">B/L</td><td>${esc(s.bl?.blNo||'')}</td></tr></table><table class="tbl"><thead><tr><th>Marks / Packages</th><th>Description of Goods</th><th>Gross Weight</th><th>Net Weight</th><th>Country of Origin</th></tr></thead><tbody><tr><td>${esc(s.customs?.marks||'')}<br>${w.bags.toLocaleString()} bags</td><td>${esc(x.description||c.product)}</td><td>${w.grossKg.toFixed(3)} kg</td><td>${w.netKg.toFixed(3)} kg</td><td>PAKISTAN</td></tr></tbody></table><div class="greenBox">GOODS OF PAKISTAN ORIGIN</div>${sigHTML(c.seller)}`,c.seller)}
function printKCCIOverlay(s,c){const x=s.coo,w=customsWeightSummary(s,c);printEngine(`KCCI COO Overlay ${x.reference||s.contractRef}`,`<div class="copyLabel">KCCI PHYSICAL FORM OVERLAY / POSITIONING DRAFT</div><h1 class="docTitle">CERTIFICATE OF ORIGIN</h1><table class="meta"><tr><td class="lbl">Exporter</td><td>${esc(SELLERS[c.seller]?.name||c.seller)}<br>${esc(c.sellerDetails?.address||'')}</td><td class="lbl">Membership</td><td>${esc(SELLERS[c.seller]?.kcci||'')}</td></tr><tr><td class="lbl">Consignee / Importer</td><td colspan="3">${esc(x.consignee||'')}</td></tr><tr><td class="lbl">Transport</td><td>${esc(s.bl?.vessel||'')} ${esc(s.bl?.voyage||'')}</td><td class="lbl">B/L</td><td>${esc(s.bl?.blNo||'')}</td></tr></table><table class="tbl"><tr><th>Marks / No. & kind of packages</th><th>Description of Goods</th><th>Gross / Net Weight</th><th>Country of Origin</th></tr><tr><td>${esc(s.customs?.marks||'')}<br>${w.bags.toLocaleString()} bags</td><td>${esc(x.description||c.product)}</td><td>Gross ${w.grossKg.toFixed(3)} kg<br>Net ${w.netKg.toFixed(3)} kg</td><td>Pakistan</td></tr></table><p>It is hereby declared that the above mentioned goods originate in Pakistan.</p>${sigHTML(c.seller)}`,c.seller)}
function certificateBody(type,s,c){const w=customsWeightSummary(s,c),goods=esc(s.customs?.description||c.product);if(type==='Phytosanitary Certificate')return`<p>This draft certifies the shipment particulars for phytosanitary processing.</p><table class="meta"><tr><td class="lbl">Goods</td><td>${goods}</td><td class="lbl">Quantity</td><td>${(w.netKg/1000).toFixed(3)} MT</td></tr><tr><td class="lbl">Origin</td><td>Pakistan</td><td class="lbl">Destination</td><td>${esc(c.destPort||c.podPort)} ${esc(c.destCountry||c.podCountry)}</td></tr></table>`;if(type==='Fumigation Certificate')return`<p>Fumigation certificate draft for the goods described below, subject to fumigator's actual treatment particulars.</p><table class="meta"><tr><td class="lbl">Goods</td><td>${goods}</td><td class="lbl">Packages</td><td>${w.bags.toLocaleString()}</td></tr></table>`;if(type==='Certificate of Weight / Quality / Packing')return`<table class="meta"><tr><td class="lbl">Goods</td><td>${goods}</td><td class="lbl">Quality</td><td>${esc(c.quality||DEFAULT_QUALITY)}</td></tr><tr><td class="lbl">Net Weight</td><td>${w.netKg.toFixed(3)} kg</td><td class="lbl">Gross Weight</td><td>${w.grossKg.toFixed(3)} kg</td></tr><tr><td class="lbl">Packages</td><td colspan="3">${w.bags.toLocaleString()} bags</td></tr></table>`;if(type==='Health Certificate')return`<p>The goods described as ${goods} are represented as rice fit for human consumption and free from live insects and bad odour, subject to the applicable authority certification.</p>`;return`<p>Inspection certificate draft for ${goods}.</p><table class="meta"><tr><td class="lbl">Inspection</td><td>${esc(s.production?.inspection||c.inspection||'')}</td><td class="lbl">Quantity</td><td>${(w.netKg/1000).toFixed(3)} MT</td></tr></table>`}
function printCertificates(s,c,types=s.certs.map(x=>x.type)){const html=types.map((type,i)=>`${i?'<div class="pageBreak"></div>':''}<h1 class="docTitle">${esc(type)}</h1>${certificateBody(type,s,c)}${sigHTML(c.seller)}`).join('');printEngine(`Shipment Certificates ${s.contractRef}`,html,c.seller)}
function exchangeDraftHTML(s,c,copy){const lc=applicableLC(s),x=s.commercial,amount=commercialValue(s,c);return`<div class="copyLabel">${esc(copy)}</div><h1 class="docTitle">EXCHANGE DRAFT</h1><table class="meta"><tr><td class="lbl">Draft Date</td><td>${fmt(x.date)}</td><td class="lbl">Amount</td><td><b>${money(amount,x.currency)}</b></td></tr><tr><td class="lbl">L/C No.</td><td>${esc(lc?.lcNo||'')}</td><td class="lbl">L/C Date</td><td>${fmt(lc?.lcDate||'')}</td></tr><tr><td class="lbl">Tenor</td><td colspan="3">${lc?.paymentCode==='LC_USANCE'?`${num(lc.usanceDays)||90} DAYS FROM B/L DATE`:'AT SIGHT'}</td></tr><tr><td class="lbl">Drawn On</td><td colspan="3">${esc(lc?.issuingBank||'Issuing Bank')}</td></tr></table><p>For value received, pay against this ${esc(copy.toLowerCase())} exchange draft under the above documentary credit.</p>${sigHTML(c.seller)}`}
function printExchangeDraft(s,c){if(!applicableLC(s))return alert('No accepted L/C is linked to this shipment.');printEngine(`Exchange Draft ${s.contractRef}`,`${exchangeDraftHTML(s,c,'ORIGINAL')}<div class="pageBreak"></div>${exchangeDraftHTML(s,c,'DUPLICATE')}`,c.seller)}
function bankCoverHTML(s,c,copyLabel='BANK COPY'){const x=s.covering,refs=bankRefs(s),lc=applicableLC(s);return`<div class="copyLabel">${esc(copyLabel)}</div><h1 class="docTitle">BANK COVERING LETTER</h1><p>To,<br><b>${esc(x.bank||'')}</b></p><p><b>Subject:</b> Original shipping documents against ${lc?`L/C No. ${esc(refs.lc)}`:'export documents'}${refs.fi.length?` · FI No(s). ${esc(refs.fi.join(', '))}`:''}${refs.gd.length?` · GD No(s). ${esc(refs.gd.join(', '))}`:''}</p><p>Please find attached the following export documents for checking and onward processing / dispatch as applicable.</p><table class="tbl"><thead><tr><th>S.#</th><th>Description</th><th>Original</th><th>Copies</th></tr></thead><tbody>${x.docs.map((r,i)=>`<tr><td>${i+1}</td><td>${esc(r.name)}</td><td>${num(r.originals)}</td><td>${num(r.copies)}</td></tr>`).join('')}</tbody></table>${x.dispatched?`<div class="greenBox">Dispatched through ${esc(x.sentThrough)} · AWB / Tracking: ${esc(x.tracking)} · ${fmt(x.dispatchDate)}</div>`:''}<p>${esc(x.remarks||'')}</p>${sigHTML(c.seller)}`}
function printBankCovering(s,c){const parts=[];if(s.covering.bankCopy)parts.push(bankCoverHTML(s,c,'BANK COPY'));if(s.covering.transtradeCopy)parts.push(bankCoverHTML(s,c,'TRANSTRADE RECEIVING COPY'));printEngine(`Bank Covering ${s.contractRef}`,parts.join('<div class="pageBreak"></div>'),c.seller)}
function printFinalPack(s,c){const x=s.commercial,sections=[];sections.push(`${commercialMeta(s,c,'COMMERCIAL INVOICE')}<table class="tbl"><tr><th>Description</th><th>Actual Qty MT</th><th>Rate / MT</th><th>Value</th></tr><tr><td>${esc(s.customs?.description||c.product)}</td><td>${actualLoadedQtyMT(s).toFixed(3)}</td><td>${money(x.rate,x.currency)}</td><td>${money(commercialValue(s,c),x.currency)}</td></tr></table>${sigHTML(c.seller)}`);sections.push(`${commercialMeta(s,c,'COMMERCIAL PACKING LIST')}<table class="tbl"><tr><th>Container</th><th>Seal</th><th>Bags</th><th>Net kg</th></tr>${uniqueActualContainers(s).map(a=>`<tr><td>${esc(a.number)}</td><td>${esc(a.seal)}</td><td>${num(a.bags).toLocaleString()}</td><td>${num(a.netKg).toLocaleString()}</td></tr>`).join('')}</table>${sigHTML(c.seller)}`);sections.push(`<h1 class="docTitle">BILL OF LADING PARTICULARS</h1><p><b>B/L:</b> ${esc(s.bl.blNo)} · <b>Vessel/Voyage:</b> ${esc(s.bl.vessel)} / ${esc(s.bl.voyage)}</p><table class="tbl"><tr><th>Container</th><th>Seal</th></tr>${uniqueActualContainers(s).map(a=>`<tr><td>${esc(a.number)}</td><td>${esc(a.seal)}</td></tr>`).join('')}</table>`);sections.push(`<h1 class="docTitle">CERTIFICATE OF ORIGIN</h1><p>${esc(s.coo.description||c.product)}</p><p>Country of Origin: <b>PAKISTAN</b></p>${sigHTML(c.seller)}`);for(const cert of s.certs||[])sections.push(`<h1 class="docTitle">${esc(cert.type)}</h1>${certificateBody(cert.type,s,c)}${sigHTML(c.seller)}`);if(s.lcId){sections.push(exchangeDraftHTML(s,c,'ORIGINAL'));sections.push(exchangeDraftHTML(s,c,'DUPLICATE'))}sections.push(bankCoverHTML(s,c,'BANK COPY'));printEngine(`Final Export Pack ${s.contractRef}`,sections.map((z,i)=>`${i?'<div class="pageBreak"></div>':''}${z}`).join(''),c.seller)}
