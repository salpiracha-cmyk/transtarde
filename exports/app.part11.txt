
function latestBagLine(s,pk,index){for(let i=s.bagOrders.length-1;i>=0;i--){const po=s.bagOrders[i],line=(po.lines||[]).find(l=>num(l.packingIndex)===index)||(po.lines||[]).find(l=>String(l.brand||'').toUpperCase()===String(pk.brand||'').toUpperCase());if(line)return line}return null}
function customsActualForPacking(s,c,pk,index){let rows=(s.millActuals||[]).filter(a=>String(a.brand||'').toUpperCase()===String(pk.brand||'').toUpperCase());if(!rows.length&&c.packings.length===1)rows=s.millActuals||[];return{rows,bags:rows.reduce((a,x)=>a+num(x.bags),0),netKg:rows.reduce((a,x)=>a+num(x.netKg),0)}}
function tareInfo(s,c,pk,index){const bo=latestBagLine(s,pk,index),contractTare=num(pk.tare),bagTare=bo?num(bo.tare):contractTare,same=Math.abs(contractTare-bagTare)<0.001,key=String(index);s.customs.tareOverrides=s.customs.tareOverrides||{};const customsTare=s.customs.tareOverrides[key]!==undefined?num(s.customs.tareOverrides[key]):contractTare;const master=bo?.masterBag||pk.masterBag||{enabled:false,qty:0,tare:0};return{contractTare,bagTare,same,customsTare,master}}
function customsWeightSummary(s,c){let netKg=0,tareKg=0,grossKg=0,bags=0;const lines=(c.packings||[]).map((pk,i)=>{const actual=customsActualForPacking(s,c,pk,i),ti=tareInfo(s,c,pk,i),masterKg=ti.master?.enabled?num(ti.master.qty)*num(ti.master.tare)/1000:0,lineTare=actual.bags*ti.customsTare/1000+masterKg,lineGross=actual.netKg+lineTare;netKg+=actual.netKg;tareKg+=lineTare;grossKg+=lineGross;bags+=actual.bags;return{index:i,pk,actual,ti,masterKg,tareKg:lineTare,grossKg:lineGross}});return{lines,netKg,tareKg,grossKg,bags}}
function customsInvoiceValue(s,c){const w=customsWeightSummary(s,c),rate=num(s.customs.rate);return round(w.netKg/1000*rate,2)}
function fiAllocationTotal(s){return(s.customs.fiAllocations||[]).reduce((a,x)=>a+num(x.amount),0)}
function settlementTotal(s){return round(fiAllocationTotal(s)+num(s.customs.bankAmount)+num(s.customs.openAccount),2)}
function isTGLinked(s,c){return!!(s.customs.tgLinked||c.seller==='TG'||/TRANS\s*GRAINS/i.test(c.buyerDetails?.name||s.buyer||''))}
function ensureCustoms(s,c){s.customs=Object.assign({saved:false,date:TODAY(),invoiceNo:`CI-${String(s.contractRef).replace(/\W/g,'-')}`,exporter:c.seller==='TG'?'TTI':c.seller,buyerImporter:c.buyerDetails?.name||s.buyer||'',buyerAddress:c.buyerDetails?.address||'',currency:c.currency||'USD',rate:0,tgLinked:/TRANS\s*GRAINS/i.test(c.buyerDetails?.name||s.buyer||''),tgInternalRate:0,fiAllocations:[],gdRefs:[],bank:'',iban:'',bankAmount:0,openAccount:0,hsCode:'',marks:'',description:c.product||'',tareOverrides:{}},s.customs||{});s.customs.fiAllocations=Array.isArray(s.customs.fiAllocations)?s.customs.fiAllocations:[];s.customs.gdRefs=Array.isArray(s.customs.gdRefs)?s.customs.gdRefs:[];s.customs.tareOverrides=s.customs.tareOverrides||{};return s.customs}

function captureCustoms(){const s=shipment(),c=contractByRef(s.contractRef),x=ensureCustoms(s,c);x.invoiceNo=byId('cuNo')?.value.trim()||x.invoiceNo;x.date=byId('cuDate')?.value||x.date;x.exporter=byId('cuExporter')?.value||x.exporter;x.currency=byId('cuCurrency')?.value.trim().toUpperCase()||x.currency;x.buyerImporter=byId('cuBuyer')?.value.trim()||'';x.buyerAddress=byId('cuBuyerAddress')?.value.trim()||'';x.hsCode=byId('cuHS')?.value.trim()||'';x.marks=byId('cuMarks')?.value.trim()||'';x.description=byId('cuDescription')?.value.trim()||'';x.tgLinked=!!byId('cuTGLinked')?.checked;if(x.tgLinked)x.tgInternalRate=num(byId('cuRate')?.value);else x.rate=num(byId('cuRate')?.value);x.bank=byId('cuBank')?.value.trim()||'';x.iban=byId('cuIBAN')?.value.trim()||'';x.bankAmount=num(byId('cuBankAmount')?.value);x.openAccount=num(byId('cuOpen')?.value);x.gdRefs=(byId('cuGD')?.value||'').split('\n').map(v=>v.trim()).filter(Boolean);document.querySelectorAll('[data-fi-no]').forEach(el=>x.fiAllocations[num(el.dataset.fiNo)].fiNo=el.value.trim());document.querySelectorAll('[data-fi-amount]').forEach(el=>x.fiAllocations[num(el.dataset.fiAmount)].amount=num(el.value));document.querySelectorAll('[data-cu-tare]').forEach(el=>x.tareOverrides[String(el.dataset.cuTare)]=num(el.value));x.rate=x.tgLinked?num(x.tgInternalRate):num(x.rate);return x}
function wireCustoms(){const s=shipment(),c=contractByRef(s.contractRef),x=ensureCustoms(s,c);for(const id of ['cuNo','cuDate','cuExporter','cuCurrency','cuBuyer','cuBuyerAddress','cuHS','cuMarks','cuDescription','cuRate','cuBank','cuIBAN','cuBankAmount','cuOpen','cuGD'])byId(id)?.addEventListener('change',()=>{captureCustoms();renderShipmentWorkspace()});byId('cuTGLinked').onchange=()=>{captureCustoms();if(x.tgLinked){x.rate=0;x.tgInternalRate=0}renderShipmentWorkspace()};byId('addFIAllocation').onclick=()=>{captureCustoms();x.fiAllocations.push({fiNo:'',amount:0});renderShipmentWorkspace()};document.querySelectorAll('[data-fi-remove]').forEach(el=>el.onclick=()=>{captureCustoms();x.fiAllocations.splice(num(el.dataset.fiRemove),1);renderShipmentWorkspace()});byId('saveCustoms').onclick=saveCustoms;byId('printCustomInvoice').onclick=()=>printCustomInvoice(s,c);byId('printCustomPacking').onclick=()=>printCustomPackingList(s,c);byId('printPhyto').onclick=()=>printPhytosanitaryInvoice(s,c)}
function saveCustoms(){const s=shipment(),c=contractByRef(s.contractRef),x=captureCustoms();if(!(s.millActuals||[]).length)return alert('Mill actual loading data is required before final Customs documents.');if(x.tgLinked&&num(x.tgInternalRate)<=0)return alert('Enter the reduced Pakistan-side intercompany/customs rate supplied by Salman/directors to Jazib. The TG customer selling price will not be inferred.');if(num(x.rate)<=0)return alert('Customs Unit Rate is required.');if((x.fiAllocations||[]).some(a=>!a.fiNo&&num(a.amount)>0))return alert('Enter FI Number for every FI allocation amount.');if((x.fiAllocations||[]).some(a=>a.fiNo&&num(a.amount)<=0))return alert('Enter an allocation amount for every FI Number.');if(num(x.bankAmount)>0&&(!x.bank||!x.iban))return alert('Bank Name and IBAN are required when Bank/IBAN settlement amount is used.');const inv=customsInvoiceValue(s,c),allocated=settlementTotal(s);if(Math.abs(inv-allocated)>0.01)return alert(`Settlement is not reconciled. Custom Invoice ${money(inv,x.currency)} versus allocated ${money(allocated,x.currency)}.`);x.saved=true;x.savedAt=new Date().toISOString();x.actualWeightSnapshot=customsWeightSummary(s,c);s.status='Customs Documents Prepared';audit('CUSTOMS','Saved controlling Custom Invoice',`${x.invoiceNo} · ${money(inv,x.currency)}`);save();renderShipmentWorkspace()}
