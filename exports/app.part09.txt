
function bagRequiredBags(p,c){const unitKg=(p.unit||c.packingUnit)==='LB'?num(p.size)*0.45359237:num(p.size);return unitKg>0?Math.ceil(num(p.containers)*num(p.weightPer)*1000/unitKg):0}
function newBagOrderDraft(s,c){const seq=s.bagOrders.length+1;return{poNo:`PO-${String(new Date().getFullYear()).slice(-2)}${String(seq).padStart(4,'0')}`,supplier:'',orderDate:TODAY(),deliveryTBA:true,requiredDate:'',deliverTo:'To Be Advised Later',lines:(c.packings||[]).map((p,i)=>({packingIndex:i,brand:p.brand||'',type:p.type||'PP Bags',size:num(p.size),unit:p.unit||c.packingUnit||'KG',tare:num(p.tare),requiredBags:bagRequiredBags(p,c),extraPct:0,extraBags:0,totalBags:bagRequiredBags(p,c),handle:'No',masterBag:clone(p.masterBag||{enabled:false,qty:0,tare:0}),artworkName:'',artworkType:'',artworkData:'',artworkAttached:false})),versions:[]}}
function ensureBagDraft(){const s=shipment(),c=contractByRef(s.contractRef);if(!s.bagOrderDraft)s.bagOrderDraft=newBagOrderDraft(s,c);return s.bagOrderDraft}
function bagOrderTotals(po){return(po.lines||[]).reduce((a,l)=>({required:a.required+num(l.requiredBags),extra:a.extra+num(l.extraBags),total:a.total+num(l.totalBags)}),{required:0,extra:0,total:0})}

function recalcBagLine(po,i){const l=po.lines[i];l.extraPct=Math.max(0,num(l.extraPct));l.extraBags=Math.ceil(num(l.requiredBags)*l.extraPct/100);l.totalBags=num(l.requiredBags)+l.extraBags}
