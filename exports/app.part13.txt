function applicableLC(s){return state.lettersOfCredit.find(x=>x.id===s.lcId)||null}
function bankRefs(s){return{fi:(s.customs?.fiAllocations||[]).map(x=>x.fiNo).filter(Boolean),gd:s.commercial?.gdRefs?.length?s.commercial.gdRefs:(s.customs?.gdRefs||[]),lc:applicableLC(s)?.lcNo||''}}
function ensureCover(s,c){s.covering=Object.assign({saved:false,dispatched:false,dispatchDate:'',sentThrough:'DHL',tracking:'',bank:'',remarks:'',docs:[],bankCopy:true,transtradeCopy:true},s.covering||{});if(!s.covering.docs.length)s.covering.docs=(c.docs||[]).map(d=>({name:Array.isArray(d)?d[0]:d.name||'',originals:num(Array.isArray(d)?d[1]:d.orig),copies:num(Array.isArray(d)?d[2]:d.copies)}));return s.covering}

class="btn green big" id="dispatchCover">SAVE DISPATCH / AWB</button></div></div>`;wireCover()}
function captureCover(){const s=shipment(),c=contractByRef(s.contractRef),x=ensureCover(s,c);x.bank=byId('coverBank').value.trim();x.sentThrough=byId('coverCourier').value;x.dispatchDate=byId('coverDate').value;x.tracking=byId('coverTracking').value.trim();x.remarks=byId('coverRemarks').value.trim();x.bankCopy=!!byId('coverBankCopy').checked;x.transtradeCopy=!!byId('coverTTICopy').checked;document.querySelectorAll('[data-cover-doc]').forEach(el=>x.docs[num(el.dataset.coverDoc)].name=el.value.trim());document.querySelectorAll('[data-cover-orig]').forEach(el=>x.docs[num(el.dataset.coverOrig)].originals=num(el.value));document.querySelectorAll('[data-cover-copy]').forEach(el=>x.docs[num(el.dataset.coverCopy)].copies=num(el.value));return x}
function wireCover(){const s=shipment(),c=contractByRef(s.contractRef),x=ensureCover(s,c);byId('saveCover').onclick=()=>{captureCover();if(!x.bank)return alert('Presenting Bank is required.');x.saved=true;x.savedAt=new Date().toISOString();audit('BANK COVERING','Prepared',s.contractRef);save();renderShipmentWorkspace()};byId('dispatchCover').onclick=()=>{captureCover();if(!x.bank)return alert('Presenting Bank is required.');if(!x.dispatchDate||!x.tracking)return alert('Dispatch Date and AWB / Tracking Number are required.');x.saved=true;x.dispatched=true;x.dispatchedAt=new Date().toISOString();audit('DISPATCH',`${x.sentThrough} documents dispatched`,x.tracking);save();renderShipmentWorkspace()};byId('printCover').onclick=()=>printBankCovering(s,c);byId('printExchange')?.addEventListener('click',()=>printExchangeDraft(s,c))}

function finalPackChecks(s,c){const z=blCompleteness(s,c);return[{label:'Accepted Sales Contract',ok:!!c.received},{label:'BAG ORDER issued',ok:s.bagOrders.length>0},{label:'Production Instructions sent',ok:!!s.production?.sentToMill},{label:'Loading lot(s) created',ok:(s.loading?.lots?.length||0)>0},{label:'Mill containers / seals complete',ok:z.complete},{label:'Customs saved',ok:!!s.customs?.saved},{label:'B/L finalized',ok:!!s.bl?.finalized},{label:'Commercial finalized',ok:!!s.commercial?.finalized},{label:'COO prepared',ok:!!s.coo?.saved},{label:'Bank Covering prepared',ok:!!s.covering?.saved}]}
function renderDocumentOutput(d){const s=shipment(),c=contractByRef(s.contractRef),checks=finalPackChecks(s,c),ok=checks.every(x=>x.ok);d.innerHTML=`<div class="section"><h3>FINAL DOCUMENTS</h3><div class="notice info">Only applicable shipment documents are assembled. BANK COVERING is always last.</div><div class="checklist">${checks.map(x=>`<div class="checkItem"><span class="checkDot ${x.ok?'ok':'bad'}"></span>${esc(x.label)}</div>`).join('')}</div><div class="section"><b>Final sequence</b><ol><li>Commercial Invoice</li><li>Commercial Packing List</li><li>B/L Particulars</li><li>Certificate of Origin</li>${(s.certs||[]).map(x=>`<li>${esc(x.type)}</li>`).join('')}${s.lcId?'<li>Exchange Draft — Original</li><li>Exchange Draft — Duplicate</li>':''}<li><b>BANK COVERING — LAST</b></li></ol></div><div class="toolbar"><div class="spacer"></div><button class="btn navy" id="printFinalPack" ${ok?'':'disabled'}>PRINT FINAL DOCUMENT PACK</button><button class="btn green" id="completeShipment" ${ok&&s.covering?.dispatched?'':'disabled'}>MARK SHIPMENT COMPLETED</button></div></div>`;byId('printFinalPack').onclick=()=>printFinalPack(s,c);byId('completeShipment').onclick=()=>{if(!finalPackChecks(s,c).every(x=>x.ok)||!s.covering?.dispatched)return alert('Complete the final document and dispatch requirements first.');s.completed=true;s.status='Completed';s.completedAt=new Date().toISOString();audit('SHIPMENT','Completed',s.contractRef);save();view='completed';currentShipmentId='';activeWorkspace='';renderCompleted()}}

function fiUsed(no){const key=String(no||'').toUpperCase();return state.shipments.reduce((a,s)=>a+(s.customs?.fiAllocations||[]).filter(x=>String(x.fiNo||'').toUpperCase()===key).reduce((z,x)=>z+num(x.amount),0),0)}
